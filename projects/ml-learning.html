NumPy Array Work Example

import numpy as np

# Feature matrix – rows are samples, columns are features
X = np.array([
    [1.0, 2.3, 0.7],
    [3.1, 1.8, 4.5],
    [0.5, 2.9, 1.1],
])

# Mean-centre each feature column (vectorised – no Python loops)
X_centered = X - X.mean(axis=0)

# Correlation matrix in one pass
corr = np.corrcoef(X_centered.T)

# Visualise as a heatmap
plt.imshow(corr, cmap="coolwarm", vmin=-1, vmax=1)
plt.colorbar(label="Correlation")
plt.title("Feature Correlation Matrix")
plt.show()
